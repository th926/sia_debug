cmake_minimum_required(VERSION 3.10)
project(SIAXX)
set(exename SIA)

set(CMAKE_CXX_FLAGS -g)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_COMPILER /opt/homebrew/opt/llvm/bin/clang++)

set(SOURCES
    Database.cpp
    wpdb.cpp
    WPConfig.cpp
    main.cpp)

add_executable(${exename} ${SOURCES})

find_package(OpenSSL REQUIRED HINTS /opt/homebrew/opt/openssl)

target_include_directories(${exename} PRIVATE ./externals/mysql-9.7.0-macos15-arm64/include)
target_link_directories(${exename} PRIVATE ./externals/mysql-9.7.0-macos15-arm64/lib)
target_link_libraries(${exename} PRIVATE libmysqlclient.a)
